# 1. Chercher les fichiers de logs
find . -name "*.log" -type f 2>/dev/null

# 2. Vérifier les logs dans le répertoire logs/
ls -la logs/ 2>/dev/null

# 3. Vérifier les logs Apache/Nginx (si accessible)
ls -la /var/log/apache2/ 2>/dev/null
ls -la /var/log/nginx/ 2>/dev/null

# 4. Vérifier les logs Passenger
ls -la log/ 2>/dev/null
cat log/passenger.log 2>/dev/null | tail -50

# 5. Créer le fichier debug.log s'il n'existe pas
touch debug.log
chmod 666 debug.log

python manage.py shell
